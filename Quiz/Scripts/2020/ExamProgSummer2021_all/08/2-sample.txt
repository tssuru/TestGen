def f(arg, n):
    n = 6
    tmp = arg
    tmp[:2] = 'bd'
    return len(tmp)>3

try:
    d = ['0','1','2','3','4']
    n = 7
    f(d, n)
    print(n, end=';')
    for el in d:
        print(el, sep=';')
except:
    print('ERR')
