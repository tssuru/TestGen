class A:
    c = 1
    b = 2

    def __init__(self):
        c = 3

class B(A):
    b = 4

    def __init__(self):
        super().__init__()
        self.a = 7

obj = B()
obj.c = obj.a + 10
B.c = 13

try: print(obj.c, end= ' ')
except: print('ERR', end=' ')

try: print(A.c, end= ' ')
except: print('ERR', end=' ')

try: print(B.c, end= ' ')
except: print('ERR', end=' ')
