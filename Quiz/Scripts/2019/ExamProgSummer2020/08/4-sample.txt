def f(arg, n):
    arg[87] = 1
    n *= -2

try:
    n = 3
    t = {46:2, 32:4, 86:2, 32:1}
    f(t, n)
    print(n, end=';')
    for x in t.items():
        print(*x, sep=';', end=';')
except:
    print('ERR')
