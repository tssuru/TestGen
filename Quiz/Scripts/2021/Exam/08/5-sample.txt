def f(arg, n):
    arg[41] = 1
    n = 6
    arg = tuple(arg.items())

try:
    n = 1
    t = {18:8, 85:3, 23:8, 73:1, 85:1}
    f(t, n)
    print(n, end=';')
    for x in t.keys():
        print(x, sep=';', end=';')
except:
    print('ERR')
