\end{verbatim}

codes = set()
regex = re.compile(r'XXX')
with open('1.txt', encoding='utf8') as f:

        res = re.fullmatch(regex, s)
        if not res: continue
        s = ''
        codes.add(YYY)
\end{verbatim}
